cmake_minimum_required(VERSION 3.18)

############ User config ############

set(THIRD_PARTY_ROOT thirdparty)
set(TOOLCHAIN_FILE arm_none_eabi.cmake)

# Fixed include sequence
include(src/bsp/bsp.cmake)
include(src/main.cmake)

# Toolchain file must be included after CONFIG and BSP selected,
# but before 'project' directive
include(${TOOLCHAIN_FILE}) # Set toolchain

set(PROJECT_OUTPUT_NAME "ecmo")
project(${PROJECT_OUTPUT_NAME}.elf LANGUAGES C)

# Generate compile_commands.json
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)


############ FreeRTOS kernel ############

set(FREERTOS_PORT GCC_ARM_CM7 CACHE STRING "")

add_library(freertos_config INTERFACE)
target_include_directories(freertos_config SYSTEM INTERFACE src/freertos_config)
target_compile_definitions(freertos_config INTERFACE projCOVERAGE_TEST=0)

add_subdirectory(${THIRD_PARTY_ROOT}/freertos)
set(EXTERNAL_LIBS ${EXTERNAL_LIBS} freertos_kernel)


############ LVGL ############
# Specify path to own LVGL config header
set(LV_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/src/bsp/include/lv_conf.h CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS    ON)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON)
add_subdirectory(${THIRD_PARTY_ROOT}/lvgl)
set(EXTERNAL_LIBS ${EXTERNAL_LIBS} lvgl)


####### SquareLine Studio UI lib #######

# Generated code with generaged CMakeLists.txt
add_subdirectory(src/hmi/codegen)

# Make LVGL includes visible to the 'ui' library generated by SquareLine
target_include_directories(ui PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/${THIRD_PARTY_ROOT}/lvgl
    ${CMAKE_CURRENT_SOURCE_DIR}/${THIRD_PARTY_ROOT}/lvgl/src
)

# Link LVGL to the UI library so lvgl symbols resolve there too
target_link_libraries(ui PUBLIC lvgl)
set(EXTERNAL_LIBS ${EXTERNAL_LIBS} ui)


############ Collect sources ############
set(C_SOURCES       ${CONFIG_C_SOURCES}   ${BSP_C_SOURCES})
set(ASM_SOURCES     ${CONFIG_ASM_SOURCES} ${BSP_ASM_SOURCES})
set(TARGET_INCLUDES ${CONFIG_INCLUDES}    ${BSP_INCLUDES})
set(SPEC_SRC_DEFS  "${CONFIG_DEFINES}     ${BSP_DEFINES}")
set_source_files_properties(${C_SOURCES}
    PROPERTIES COMPILE_FLAGS ${SPEC_SRC_DEFS}
)

# enable asm for stm startup.s file
enable_language(ASM)

add_executable(${PROJECT_NAME}
    ${C_SOURCES}
    ${ASM_SOURCES}
    ${EXTERNAL_EXECUTABLES}
)

############ Build target ############

# Specific 'config' headers
target_include_directories(${PROJECT_NAME} PUBLIC ${TARGET_INCLUDES})

if (EXTERNAL_DEPENDENCIES)
    add_dependencies(${PROJECT_OUTPUT_NAME}.elf
        ${EXTERNAL_DEPENDENCIES}
    )
endif()

if (EXTERNAL_LIBS)
    target_link_libraries(${PROJECT_OUTPUT_NAME}.elf
        ${EXTERNAL_LIBS}
    )
endif()

add_custom_target ( ${PROJECT_OUTPUT_NAME}.hex ALL
    DEPENDS ${PROJECT_OUTPUT_NAME}.elf 
    COMMAND ${CMAKE_OBJCOPY} -O ihex ${PROJECT_OUTPUT_NAME}.elf ${PROJECT_OUTPUT_NAME}.hex
    COMMENT "Generating ${PROJECT_OUTPUT_NAME}.hex"
    )

add_custom_command(TARGET ${PROJECT_OUTPUT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_SIZE} --format=berkeley "${PROJECT_OUTPUT_NAME}.elf")
